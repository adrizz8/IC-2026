# 1. Ve a la carpeta de la nueva práctica
cd ~/Universidad/Cuarto/Ingenieria\ del\ Conocimiento/IC-2026/Practica\ 3

# 2. Crea el entorno virtual (siempre con Python 3.12)
/opt/homebrew/bin/python3.12 -m venv venv

source venv/bin/activate

python -m ipykernel install --user --name=ic-practica3 --display-name "IC Practica 3"

pip install -r requirements.txt
